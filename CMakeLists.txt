cmake_minimum_required(VERSION 3.20)
project(RyCenter VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

function(rycenter_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

include(CTest)
if(BUILD_TESTING)
    add_executable(geometry_tests tests/geometry_tests.cpp)
    target_include_directories(geometry_tests PRIVATE src)
    rycenter_warnings(geometry_tests)
    add_test(NAME geometry COMMAND geometry_tests)
    add_executable(shortcut_tests tests/shortcut_tests.cpp)
    target_include_directories(shortcut_tests PRIVATE src)
    rycenter_warnings(shortcut_tests)
    add_test(NAME shortcuts COMMAND shortcut_tests)
endif()

if(WIN32)
    enable_language(RC)
    add_executable(RyCenter WIN32 src/main.cpp src/settings_ui.cpp resources/app.rc)
    target_include_directories(RyCenter PRIVATE resources)
    target_compile_definitions(RyCenter PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(RyCenter PRIVATE user32 shell32 dwmapi advapi32 comctl32 gdi32 gdiplus)
    rycenter_warnings(RyCenter)
    if(MSVC)
        target_compile_options(RyCenter PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL>)
        target_link_options(RyCenter PRIVATE /MANIFEST:NO $<$<CONFIG:Release>:/LTCG> /OPT:REF /OPT:ICF)
    else()
        target_compile_options(RyCenter PRIVATE -Os -ffunction-sections -fdata-sections)
        target_link_options(RyCenter PRIVATE -municode -static -Wl,--gc-sections $<$<CONFIG:Release>:-s>)
    endif()
    if(BUILD_TESTING)
        add_executable(windows_tests tests/windows_tests.cpp src/settings_ui.cpp resources/app.rc)
        target_include_directories(windows_tests PRIVATE resources)
        target_compile_definitions(windows_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
        target_link_libraries(windows_tests PRIVATE user32 shell32 dwmapi advapi32 comctl32 gdi32 gdiplus)
        rycenter_warnings(windows_tests)
        if(MSVC)
            target_link_options(windows_tests PRIVATE /MANIFEST:NO)
        endif()
        add_test(NAME windows_integration COMMAND windows_tests)
        set_tests_properties(windows_integration PROPERTIES TIMEOUT 60)
    endif()
endif()
